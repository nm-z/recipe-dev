+const EDGE_SMALL_BYTES: usize = 1 << 12;
+const EDGE_LARGE_BYTES: usize = 1 << 20;
+fn measure_edge(gpu: &'static Gpu) -> Result<DeviceEdge> {
+	let mut staging = vec![0_u8; EDGE_LARGE_BYTES];
+	let target = gpu.allocate(EDGE_LARGE_BYTES)?;
+	gpu.upload(target, staging.as_ptr().cast(), EDGE_SMALL_BYTES)?;
+	gpu.synchronize()?;
+	let mut timed = |up: bool, bytes: usize| -> Result<f64> {
+		let started = Instant::now();
+		if up {
+			gpu.upload(target, staging.as_ptr().cast(), bytes)?;
+		} else {
+			gpu.download(staging.as_mut_ptr().cast(), target, bytes)?;
+		}
+		gpu.synchronize()?;
+		Ok(started.elapsed().as_secs_f64())
+	};
+	let mut directions = [(0.0, 0.0); 2];
+	for (index, up) in [true, false].into_iter().enumerate() {
+		let small = timed(up, EDGE_SMALL_BYTES)?;
+		let large = timed(up, EDGE_LARGE_BYTES)?;
+		// The slope of the two transfers is bandwidth, the intercept latency.
+		let bandwidth = (EDGE_LARGE_BYTES - EDGE_SMALL_BYTES) as f64 / (large - small).max(f64::MIN_POSITIVE);
+		directions[index] = ((small - EDGE_SMALL_BYTES as f64 / bandwidth).max(0.0), bandwidth);
+	}
+	gpu.free(target);
+	Ok(DeviceEdge { up_latency: directions[0].0, up_bandwidth: directions[0].1, down_latency: directions[1].0, down_bandwidth: directions[1].1 })

+/// The planned work of one complete fused epoch: every contraction's forward,
+/// gradient, and previous-adjoint tile, every node's elementwise traffic, the
+/// loss reduction, and the optimizer update, in one unit the calibration
+/// measures per device.
+fn planned_epoch_work(graph: &Graph, rows: usize) -> Result<f64> {
+	let mut work = 0.0;
+	for shape in native_contraction_shapes(graph, rows)?.iter().flatten() {
+		for extent in [shape.forward, shape.gradient, shape.previous] {
+			work += 2.0 * f64::from(extent.m) * f64::from(extent.n) * f64::from(extent.k);
+		}
+	}
+	for node in &graph.nodes {
+		work += 8.0 * rows as f64 * node.output.elements() as f64;
+	}
+	work += 16.0 * graph.parameters.len() as f64;
+	work += 8.0 * (rows * graph.output.elements()) as f64;
+	Ok(work)

+const CALIBRATION_FEATURES: usize = 24;

+/// The complete predicted epoch of one candidate route: concurrent shard
+/// computation at each device's measured rate, plus the priced movement list
+/// the executor performs — statistic and gradient transfers, one crossing
+/// latency and one dispatch overhead per synchronization, and the separate
+/// persistent weight, moment, and variance movements.
+fn route_cost(topology: &Topology, route: &[usize], graph: &Graph, rows: usize, precision: NativePrecision) -> Result<RouteCost> {
+	let shards = shard_rows(rows, route.len());
+	let (mut computation, mut synchronization) = (0.0_f64, 0.0_f64);
+	for (index, device) in route.iter().enumerate() {
+		let rate = &topology.rates[*device];
+		computation = computation.max(planned_epoch_work(graph, shards[index])? / rate.rate);
+		synchronization += rate.overhead;
+	}
+	if route.len() > 1 {
+		// The aggregated update is one more dispatch on the leading device.
+		let primary = &topology.rates[route[0]];
+		computation += 16.0 * graph.parameters.len() as f64 / primary.rate;
+		synchronization += primary.overhead;
+	}
+	let (mut transfers, mut movement) = (0.0, 0.0);
+	for entry in route_movement(route.len(), graph.parameters.len(), precision) {
+		let edge = &topology.edges[route[entry.shard]];
+		let (latency, bandwidth) = if entry.upload { (edge.up_latency, edge.up_bandwidth) } else { (edge.down_latency, edge.down_bandwidth) };
+		synchronization += latency;
+		let seconds = entry.bytes as f64 / bandwidth;
+		match entry.kind {
+			MovementKind::Weights | MovementKind::Moments | MovementKind::Variances => movement += seconds,
+			_ => transfers += seconds,
+		}
+	}
+	Ok(RouteCost { computation, transfers, synchronization, movement })

+				}

+				candidate.sort_by(|left, right| topology.rates[*right].rate.total_cmp(&topology.rates[*left].rate).then_with(|| topology.devices[*left].canonical().cmp(&topology.devices[*right].canonical())));
+				let cost = route_cost(topology, &candidate, graph, rows, precision)?;

+				eprintln!("route {names} predicted epoch {:.9}s = computation {:.9} + transfers {:.9} + synchronization {:.9} + persistent-state {:.9}", cost.total(), cost.computation, cost.transfers, cost.synchronization, cost.movement);

+					best = Some((candidate, cost.total()));
+				}
+			}
+			let (selected, total) = best.ok_or_else(|| RecipeError::new("no candidate route fits the workload"))?;
+			eprintln!("selected route {} predicted epoch {total:.9}s", selected.iter().map(|device| topology.devices[*device].canonical()).collect::<Vec<_>>().join(","));
+			Ok(selected.into_iter().map(|device| topology.devices[device]).collect())
+		}
+	}
